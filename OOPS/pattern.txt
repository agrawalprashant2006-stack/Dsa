import java.util.Scanner;

public class Pattern4 {
    static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int row = 1;
        while(row<=n){
            int space = n-1;
            while(space>=row){
                System.out.print("  ");
                space--;
            }
            int star = 1;
            while(star<=row){
                System.out.print("* ");
                star++;
            }
            System.out.println();
            row++;
        }

//        5
//                *
//              * *
//            * * *
//          * * * *
//        * * * * *

    }
}

import java.util.Scanner;

public class pattern8 {
    static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int row = 1;
        int space = 2*n-3;
        int star = 1;
        while(row<=n){
            //star
            int i =1;
            while(i<=star){
                System.out.print("* ");
                i++;
            }
            //space
            int j =1;
            while(j<=space){
                System.out.print("  ");
                j++;
            }
            //star
            int k=1;
            if(row==n){
                k=2;
            }
            while(k<=star){
                System.out.print("* ");
                k++;
            }
            //next line ki prep
            System.out.println();
            row++;
            space-=2;
            star++;
        }
    }
}

        int space = -1;
        while(row<=n){
            //star
            int i =1;
            while(i<=star){
                System.out.print("* ");
                i++;
            }
            //space
            int j = 1;
            while(j<=space){
                System.out.print("  ");
                j++;
            }

            //star
            int k = 1;
            if(row==1 || row==n){
                k=2;
            }
            while(k<=n){
                System.out.print("* ");
                k++;
            }

            //mirror
            if(row< n / 2 +  1){
                star--;
                space+=2;
            }else{
                star++;
                space-=2;
            }
            //next line ki prep

import java.util.Scanner;

public class pattern17 {
    static void main(String[] args) {
        Scanner sc =new Scanner(System.in);
        int n = sc.nextInt();
        int row =1;
        int space = n-1;
        int star = 1;
        int val =1;
        while(row<=2*n-1){
            //space
            int i =1;
            while(i<=space){
                System.out.print("  ");
                i++;
            }
            //star
            int j =1;
            int p = val;
            while(j<=star){
                System.out.print(p+" ");
                if(j<star/2+1){
                    p++;
                }else{
                    p--;
                }
                j++;
            }
            //mirror
            if(row<n){
                space--;
                star+=2;
                val++;
            }else{
                space++;
                star-=2;
                val--;
            }
            //next line ki prep
            row++;
            System.out.println();
        }
    }
}
